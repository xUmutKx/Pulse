cmake_minimum_required(VERSION 3.22)
project(pulsewhisper C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED WHISPER_SRC)
    message(FATAL_ERROR "WHISPER_SRC is not set (the workflow fetches whisper.cpp into localai/vendor)")
endif()
# whisper and its ggml as static libraries, folded into libpulsewhisper.so with nothing of them exported
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_subdirectory(${WHISPER_SRC} build-whisper)
add_library(pulsewhisper SHARED jni.c)
target_include_directories(pulsewhisper PRIVATE ${WHISPER_SRC}/include ${WHISPER_SRC}/ggml/include)
target_link_libraries(pulsewhisper whisper log)
target_link_options(pulsewhisper PRIVATE -Wl,--exclude-libs,ALL -Wl,--gc-sections)
target_compile_options(pulsewhisper PRIVATE -O3)
